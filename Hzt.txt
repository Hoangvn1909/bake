local function addLinkButton(parent, y, color, icon, displayText, fullUrl)
    local btnFrame = Instance.new("Frame")
    btnFrame.Size = UDim2.new(0, 440, 0, 50)
    btnFrame.Position = UDim2.new(0.5, -220, 0, y)
    btnFrame.BackgroundColor3 = color
    btnFrame.BackgroundTransparency = 0.15
    btnFrame.BorderSizePixel = 0
    btnFrame.Parent = parent
    local btnCorner = Instance.new("UICorner")
    btnCorner.CornerRadius = UDim.new(0, 12)
    btnCorner.Parent = btnFrame
    
    local textLabel = Instance.new("TextLabel")
    textLabel.Size = UDim2.new(1, -80, 1, 0)
    textLabel.Position = UDim2.new(0, 15, 0, 0)
    textLabel.BackgroundTransparency = 1
    textLabel.Text = icon .. " " .. displayText
    textLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
    textLabel.TextSize = 14
    textLabel.Font = Enum.Font.Gotham
    textLabel.TextXAlignment = Enum.TextXAlignment.Left
    textLabel.Parent = btnFrame
    
    local copyBtn = Instance.new("TextButton")
    copyBtn.Size = UDim2.new(0, 80, 1, 0)
    copyBtn.Position = UDim2.new(1, -90, 0, 0)
    copyBtn.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
    copyBtn.BackgroundTransparency = 0.3
    copyBtn.Text = "Copy"
    copyBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
    copyBtn.TextSize = 14
    copyBtn.Font = Enum.Font.GothamBold
    copyBtn.Parent = btnFrame
    local copyCorner = Instance.new("UICorner")
    copyCorner.CornerRadius = UDim.new(0, 8)
    copyCorner.Parent = copyBtn
    
    copyBtn.MouseButton1Click:Connect(function()
        setclipboard(fullUrl)
        local old = copyBtn.Text
        copyBtn.Text = "✓ Copied!"
        task.wait(1)
        copyBtn.Text = old
    end)
end

-- Timer
local timerText = Instance.new("TextLabel")
timerText.Size = UDim2.new(1, 0, 0, 50)
timerText.Position = UDim2.new(0, 0, 0, 390)
timerText.BackgroundTransparency = 1
timerText.Text = "⏰ Kicking in " .. COUNTDOWN_SECONDS .. " seconds..."
timerText.TextColor3 = Color3.fromRGB(255, 200, 100)
timerText.TextSize = 18
timerText.Font = Enum.Font.GothamBold
timerText.TextXAlignment = Enum.TextXAlignment.Center
timerText.Parent = card


local start = tick()
while true do
    local elapsed = tick() - start
    local remaining = COUNTDOWN_SECONDS - math.floor(elapsed)
    if remaining <= 0 then break end
    timerText.Text = "⏰ Kicking in " .. remaining .. " seconds..."
    runService.RenderStepped:Wait()
end
